// Google Apps Script endpoint for Lead Capture
function doPost(e) {
  // 1. Get the active sheet
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Leads");
  if (!sheet) {
    sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("Leads");
    sheet.appendRow(["Timestamp", "Name", "Phone", "Email", "Postcode", "Form Type", "Tracking CTA", "Service", "gclid", "fbclid", "msclkid"]);
  }

  try {
    var data = JSON.parse(e.postData.contents);
    var timestamp = new Date();
    
    // 2. Append to Spreadsheet
    sheet.appendRow([
      timestamp,
      data.name || "",
      data.phone || "",
      data.email || "",
      data.postcode || "",
      data.formType || "Unknown Form",
      data.trackingSource || "",
      data.service || "",
      data.gclid || "",
      data.fbclid || "",
      data.msclkid || ""
    ]);

    // 3. Send Email Notification
    var htmlBody = `
      <h2>New Lead from Curtains Direct</h2>
      <p><strong>Name:</strong> ${data.name}</p>
      <p><strong>Phone:</strong> ${data.phone}</p>
      <p><strong>Email:</strong> ${data.email}</p>
      <p><strong>Postcode:</strong> ${data.postcode}</p>
      <p><strong>Service Needed:</strong> ${data.service}</p>
      <hr>
      <p><strong>Captured From:</strong> ${data.trackingSource}</p>
      <p><em>Form Intent:</em> ${data.formType}</p>
    `;

    MailApp.sendEmail({
      // --------- EDIT YOUR EMAIL HERE ---------
      to: "sales@curtainsdirect.com.au, gerard@curtainsdirect.com.au", 
      subject: "New Lead Capture: " + data.formType,
      htmlBody: htmlBody
    });

    return ContentService.createTextOutput(JSON.stringify({ "status": "success" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Required to handle preflight CORS requests from your React app
function doOptions(e) {
  var headers = {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Methods": "POST, GET, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type",
    "Access-Control-Max-Age": "86400"
  };
  return ContentService.createTextOutput("")
    .setMimeType(ContentService.MimeType.TEXT)
    .setHeaders(headers);
}
